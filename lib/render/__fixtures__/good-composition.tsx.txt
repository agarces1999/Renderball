// Hand-written fixture mirroring real Design/Choreography Agent output
// (see lib/agents/prompts/design-agent.ts skeleton + src/generated/*).
// Stored as .txt so tsc skips it — it is compiled at test time by
// verifyScenesRender's esbuild step, exactly like production output.
import React from "react";
import { Zap } from "lucide-react";
import { Img } from "./Img";

interface Script {
  scenes: Array<{
    content: {
      eyebrow?: string;
      headline?: string;
      lede?: string;
      caption?: string;
      asset_ids?: string[];
    };
  }>;
  assets: {
    images: Array<{ id: string; src: string; alt_text?: string }>;
    fonts: Array<{ family: string }>;
  };
  config: Record<string, unknown>;
}

const PALETTE = {
  primary: "#10141C",
  accent: "#00C28A",
  light: "#EAEDF1",
};

const BRAND_FONTS_CSS = `
  @keyframes fadeRise {
    from { opacity: 0; transform: translateY(16px); }
    to { opacity: 1; transform: translateY(0); }
  }
  @keyframes scaleIn {
    0% { opacity: 0; transform: scale(0.85); }
    100% { opacity: 1; transform: scale(1); }
  }
`;

const SECTION_FRAME: React.CSSProperties = {
  position: "absolute",
  inset: 0,
  width: "100%",
  height: "100%",
  overflow: "hidden",
  boxSizing: "border-box",
};

export const Section0: React.FC<{ script: Script }> = ({ script }) => {
  const content = script.scenes[0].content;
  return (
    <div
      style={{
        ...SECTION_FRAME,
        background: `radial-gradient(120% 80% at 50% 0%, #1a2030 0%, ${PALETTE.primary} 100%)`,
        fontFamily: '"Geist", sans-serif',
      }}
    >
      <style dangerouslySetInnerHTML={{ __html: BRAND_FONTS_CSS }} />
      <div
        style={{
          position: "absolute",
          top: 320,
          left: 0,
          right: 0,
          textAlign: "center",
          animation: "fadeRise 0.8s ease both",
        }}
      >
        <h6
          style={{
            color: PALETTE.accent,
            fontSize: 22,
            letterSpacing: "0.2em",
            textTransform: "uppercase",
            margin: 0,
          }}
        >
          {content.eyebrow}
        </h6>
        <h1 style={{ color: "#ffffff", fontSize: 96, lineHeight: 1.05, margin: "24px 0 0" }}>
          {content.headline}
        </h1>
        <p style={{ color: "rgba(255,255,255,0.7)", fontSize: 28, maxWidth: 920, margin: "24px auto 0" }}>
          {content.lede}
        </p>
      </div>
    </div>
  );
};

export const Section1: React.FC<{ script: Script }> = ({ script }) => {
  const content = script.scenes[1].content;
  const asset = script.assets.images.find(
    (img) => img.id === (content.asset_ids ?? [])[0],
  );
  return (
    <div
      style={{
        ...SECTION_FRAME,
        background: PALETTE.light,
        fontFamily: '"Geist", sans-serif',
        display: "flex",
        alignItems: "center",
        padding: 120,
      }}
    >
      <div style={{ flex: 1 }}>
        <h6 style={{ color: PALETTE.accent, fontSize: 22, letterSpacing: "0.2em", margin: 0 }}>
          {content.eyebrow}
        </h6>
        <h1 style={{ color: PALETTE.primary, fontSize: 72, lineHeight: 1.1, margin: "24px 0 0" }}>
          {content.headline}
        </h1>
        <p style={{ color: PALETTE.primary, fontSize: 22, marginTop: 32 }}>
          <Zap size={20} color={PALETTE.accent} /> {content.caption}
        </p>
      </div>
      <div style={{ flex: 1, animation: "scaleIn 0.6s ease 0.4s both" }}>
        {asset ? (
          <Img
            src={asset.src}
            alt={asset.alt_text ?? ""}
            style={{ width: "100%", borderRadius: 16 }}
          />
        ) : null}
      </div>
    </div>
  );
};

export const Generated: React.FC<{ script: Script }> = ({ script }) => (
  <>
    <Section0 script={script} />
    <Section1 script={script} />
  </>
);
